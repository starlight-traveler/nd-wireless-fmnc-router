146static void process_tcp_options_client(std::shared_ptr<Client::Data> data, u_char *options, int tcp_options_length, std::vector<int> &option_kinds)
147{
148 int parsed_length = 0;
149 while (parsed_length < tcp_options_length)
150 {
151 u_char kind = options[parsed_length];
152 option_kinds.push_back(kind);
153
154 if (kind == 0) // End of options
155 {
156 parsed_length++;
157 break;
158 }
159 else if (kind == 1) // NOP
160 {
161 parsed_length++;
162 continue;
163 }
164 else
165 {
166 if (parsed_length + 1 >= tcp_options_length)
167 {
168 break; // malformed
169 }
170 u_char length = options[parsed_length + 1];
171 if (length < 2 || parsed_length + length > tcp_options_length)
172 {
173 break; // malformed
174 }
175
176 {
177 std::lock_guard<std::mutex> lock(data->counts_mutex);
178 data->tcp_option_counts[kind]++;
179 }
180
181 parsed_length += length;
182 }
183 }
184}

245{

251

254
255 // Only process IP packets

257 {
258 return;
259 }
260

262 struct iphdr *ip_header = (struct iphdr *)(buffer + sizeof(struct ethhdr));
263 // Parse TCP header if protocol == TCP
264 if (ip_header->protocol == IPPROTO_TCP)
265 {
266 int ip_header_length = ip_header->ihl * 4;
267 struct tcphdr *tcp_header = (struct tcphdr *)(buffer + sizeof(struct ethhdr) + ip_header_length);

270 process_tcp_flags_client(args->data, tcp_flags);

275 if (tcp_options_length > 0)
276 {
277 u_char *options = (u_char *)tcp_header + sizeof(struct tcphdr);
278 process_tcp_options_client(args->data, options, tcp_options_length, option_kinds);
279 }

283 std::string timestamp = get_current_timestamp();

286 ClientPacketLogEntry log_entry;
287 log_entry.status = "received_client";
288 log_entry.timestamp = timestamp;
289 log_entry.flags = get_tcp_flag_names(tcp_flags);
290 log_entry.options = option_kinds;
291 args->data->packet_log[packet_id] = log_entry;

293 }

300 return;
301 }
302
303 // Update Ethernet header
304 memcpy(eth->h_source, src_mac, 6); // Set source MAC to our interface's MAC

306

313 {

319 }

321 {
322 LOG_DEBUG(args->logger, "Forwarded packet, length: {} bytes", sent);
